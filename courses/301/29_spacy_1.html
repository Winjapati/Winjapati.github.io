def ttr(words):
    return len(set(words)) / len(words)

def ttr_first(words, n=10000):
    return len(set(words[:n])) / len(words[:n])

def avg_word_len(words):
    total = 0
    for w in words:
        total += len(w)
    return total / len(words)

def ly_rate(words):
    count = 0
    for w in words:
        if w.endswith("ly"):
            count += 1
    return count / len(words) * 1000

def stopword_rate(words, stopwords):
    count = 0
    for w in words:
        if w in stopwords:
            count += 1
    return count / len(words)

def load_text(filename):
    with open(os.path.join(HERE, "gutenberg", filename), "r", encoding="utf-8") as f:
        lines = f.readlines()
    while not lines[0].startswith("*** START"):
        lines = lines[1:]
    lines = lines[1:]
    while not lines[-1].startswith("*** END"):
        lines = lines[:-1]
    lines = lines[:-1]
    return "".join(lines)

def load_words(filename):
    text = load_text(filename)
    words = re.split(r"[\W_]+", text.lower())
    return [w for w in words if w != ""]

def report(title, words):
    print(title, "TTR:", ttr(words))
    print(title, "TTR (10,000 words):", ttr_first(words))
    print(title, "Average Word Length:", avg_word_len(words))
    print(title, "-ly words per 1,000:", ly_rate(words))
    return ttr_first(words)

def book_to_df(filename, label):
    text = load_text(filename)
    chapters = re.split(r"(?im)^CHAPTER\s+(?:[IVXLCDM]+|\d+)\b", text)
    chapters = chapters[1:]
    rows = []
    chapter_num = 0
    for chapter in chapters:
        chapter_num += 1
        words = re.split(r"[\W_]+", chapter.lower())
        words = [w for w in words if w != ""]
        for w in words:
            rows.append({"word": w, "chapter": chapter_num})
    df = pd.DataFrame(rows)
    df["text"] = label
    return df

def add_columns(df, stopwords):
    df["length"] = df["word"].str.len()
    df["is_ly"] = df["word"].str.endswith("ly")
    df["is_stopword"] = df["word"].isin(stopwords)
    return df

What We Need

What spaCy Adds

What Is spaCy?

  • A Python library for natural language processing (NLP).
  • It comes with trained models of languages.
  • For each word, a model predicts:
    • its lemma
    • its part of speech
    • (and much more, later this week)

What Each Part Does

  • Which of these jobs could spaCy do instead?
  1. Load the book
  2. Split it into chapters
  3. Get ready to loop through the chapters
  4. Split a chapter into words
  5. Make one row per word
  6. Turn the rows into a DataFrame

What Changes?

  • Only jobs 4 and 5 – what happens inside each chapter.
  • Everything else stays the same.
  • We’ll make the changes one at a time.

def book_to_spacy_df(filename, label, nlp):
    text = load_text(filename)
    chapters = re.split(r"(?im)^CHAPTER\s+(?:[IVXLCDM]+|\d+)\b", text)
    chapters = chapters[1:]
    rows = []
    chapter_num = 0
    for chapter in chapters:
        chapter_num += 1
        doc = nlp(chapter)
        for token in doc:
            if token.is_alpha:
                rows.append({"word": token.text.lower(),
                             "lemma": token.lemma_.lower(),
                             "pos": token.pos_,
                             "is_stop": token.is_stop,
                             "chapter": chapter_num})
    df = pd.DataFrame(rows)
    df["text"] = label
    return df
