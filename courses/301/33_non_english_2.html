import os
import re
import pandas as pd
from nltk.corpus import wordnet as wn

HERE = os.path.dirname(__file__)   # the folder lin301.py lives in: data/

def ttr(words):
    return len(set(words)) / len(words)

def ttr_first(words, n=10000):
    return len(set(words[:n])) / len(words[:n])

def avg_word_len(words):
    total = 0
    for w in words:
        total += len(w)
    return total / len(words)

def ly_rate(words):
    count = 0
    for w in words:
        if w.endswith("ly"):
            count += 1
    return count / len(words) * 1000

def stopword_rate(words, stopwords):
    count = 0
    for w in words:
        if w in stopwords:
            count += 1
    return count / len(words)

def load_text(filename):
    with open(os.path.join(HERE, "gutenberg", filename), "r", encoding="utf-8") as f:
        lines = f.readlines()
    while not lines[0].startswith("*** START"):
        lines = lines[1:]
    lines = lines[1:]
    while not lines[-1].startswith("*** END"):
        lines = lines[:-1]
    lines = lines[:-1]
    return "".join(lines)

def load_words(filename):
    text = load_text(filename)
    words = re.split(r"[\W_]+", text.lower())
    return [w for w in words if w != ""]

def report(title, words):
    print(title, "TTR:", ttr(words))
    print(title, "TTR (10,000 words):", ttr_first(words))
    print(title, "Average Word Length:", avg_word_len(words))
    print(title, "-ly words per 1,000:", ly_rate(words))
    return ttr_first(words)

def book_to_df(filename, label):
    text = load_text(filename)
    chapters = re.split(r"(?im)^CHAPTER\s+(?:[IVXLCDM]+|\d+)\b", text)
    chapters = chapters[1:]
    rows = []
    chapter_num = 0
    for chapter in chapters:
        chapter_num += 1
        words = re.split(r"[\W_]+", chapter.lower())
        words = [w for w in words if w != ""]
        for w in words:
            rows.append({"word": w, "chapter": chapter_num})
    df = pd.DataFrame(rows)
    df["text"] = label
    return df

def add_columns(df):
    df["length"] = df["word"].str.len()
    df["is_ly"] = df["word"].str.endswith("ly")
    return df

def book_to_spacy_df(filename, label, nlp, chapter_word="CHAPTER"):
    text = load_text(filename)
    chapters = re.split(r"(?m)^" + chapter_word + r"\b.*$", text)
    chapters = chapters[1:]
    rows = []
    chapter_num = 0
    for chapter in chapters:
        chapter_num += 1
        doc = nlp(chapter)
        sentence_num = 0
        for sent in doc.sents:
            sentence_num += 1
            for token in sent:
                if token.is_alpha:
                    rows.append({"word": token.text.lower(),
                                 "lemma": token.lemma_.lower(),
                                 "pos": token.pos_,
                                 "morph": str(token.morph),
                                 "is_stop": token.is_stop,
                                 "sentence": sentence_num,
                                 "chapter": chapter_num})
    df = pd.DataFrame(rows)
    df["text"] = label
    return df

def book_to_svo_df(filename, label, nlp):
    text = load_text(filename)
    chapters = re.split(r"(?im)^CHAPTER\s+(?:[IVXLCDM]+|\d+)\b", text)
    chapters = chapters[1:]
    rows = []
    chapter_num = 0
    for chapter in chapters:
        chapter_num += 1
        doc = nlp(chapter)
        for token in doc:
            if token.pos_ == "VERB":
                subjects = [c for c in token.children if c.dep_ == "nsubj"]
                objects = [c for c in token.children if c.dep_ == "dobj"]
                if subjects and objects:
                    rows.append({"subject": subjects[0].lemma_.lower(),
                                 "verb": token.lemma_.lower(),
                                 "object": objects[0].lemma_.lower(),
                                 "chapter": chapter_num})
    df = pd.DataFrame(rows)
    df["text"] = label
    return df

def is_a(word, category):
    for synset in wn.synsets(word, pos=wn.NOUN):
        for path in synset.hypernym_paths():
            if category in path:
                return True
    return False

Unicode Knows the Script

A Function: script_of()

def script_of(char):
    return unicodedata.name(char).split()[0]

print(script_of("γ"), script_of("猫"), script_of("は"))
GREEK CJK HIRAGANA
  • .split() breaks the name into words; [0] takes the first one.

Romanization

The Quick Way: unidecode

In the terminal (once):

python -m pip install unidecode pypinyin pykakasi
from unidecode import unidecode

print(unidecode("Η γάτα τρώει τη νυχτερίδα;"))
print(unidecode("猫吃蝙蝠吗？"))
print(unidecode("猫はコウモリを食べるの？"))
E gata troei te nukhterida;
Mao Chi Bian Fu Ma ?
Mao hakoumoriwoShi beruno?
  • Greek: pretty good. Chinese: no tones. Japanese: wrong!

Why Is the Japanese Wrong?

  • unidecode reads every CJK character the Chinese way: 猫 → Mao, 食 → Shi.
  • In Japanese, 猫 is neko and 食べる is taberu.
  • And it doesn’t know where words begin and end: hakoumoriwo
  • Romanization is language-specific. Use a tool built for the language.

Chinese: pypinyin

from pypinyin import pinyin

syllables = pinyin("猫吃蝙蝠吗？")
print(syllables)
print(" ".join([s[0] for s in syllables]))
[['māo'], ['chī'], ['biān'], ['fú'], ['ma'], ['？']]
māo chī biān fú ma ？
  • pinyin() gives a list of lists, one per character; s[0] takes each one out.
  • The tone marks are back!

Japanese: pykakasi

import pykakasi

kks = pykakasi.kakasi()
result = kks.convert("猫はコウモリを食べるの？")
print(" ".join([item["hepburn"] for item in result]))
neko ha koumori wo tabe runo ？
  • neko and taberu: the Japanese readings
  • は and を are written ha and wo, but pronounced wa and o as particles!

def is_a(word, category, lang="eng"):
