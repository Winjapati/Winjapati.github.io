import os
import re
import pandas as pd

HERE = os.path.dirname(__file__)   # the folder lin301.py lives in: data/

def ttr(words):
    return len(set(words)) / len(words)

def ttr_first(words, n=10000):
    return len(set(words[:n])) / len(words[:n])

def avg_word_len(words):
    total = 0
    for w in words:
        total += len(w)
    return total / len(words)

def ly_rate(words):
    count = 0
    for w in words:
        if w.endswith("ly"):
            count += 1
    return count / len(words) * 1000

def stopword_rate(words, stopwords):
    count = 0
    for w in words:
        if w in stopwords:
            count += 1
    return count / len(words)

def load_text(filename):
    with open(os.path.join(HERE, "gutenberg", filename), "r", encoding="utf-8") as f:
        lines = f.readlines()
    while not lines[0].startswith("*** START"):
        lines = lines[1:]
    lines = lines[1:]
    while not lines[-1].startswith("*** END"):
        lines = lines[:-1]
    lines = lines[:-1]
    return "".join(lines)

def load_words(filename):
    text = load_text(filename)
    words = re.split(r"[\W_]+", text.lower())
    return [w for w in words if w != ""]

def report(title, words):
    print(title, "TTR:", ttr(words))
    print(title, "TTR (10,000 words):", ttr_first(words))
    print(title, "Average Word Length:", avg_word_len(words))
    print(title, "-ly words per 1,000:", ly_rate(words))
    return ttr_first(words)

def book_to_df(filename, label):
    text = load_text(filename)
    chapters = re.split(r"(?im)^CHAPTER\s+(?:[IVXLCDM]+|\d+)\b", text)
    chapters = chapters[1:]
    rows = []
    chapter_num = 0
    for chapter in chapters:
        chapter_num += 1
        words = re.split(r"[\W_]+", chapter.lower())
        words = [w for w in words if w != ""]
        for w in words:
            rows.append({"word": w, "chapter": chapter_num})
    df = pd.DataFrame(rows)
    df["text"] = label
    return df

def add_columns(df):
    df["length"] = df["word"].str.len()
    df["is_ly"] = df["word"].str.endswith("ly")
    return df

def book_to_spacy_df(filename, label, nlp):
    text = load_text(filename)
    chapters = re.split(r"(?im)^CHAPTER\s+(?:[IVXLCDM]+|\d+)\b", text)
    chapters = chapters[1:]
    rows = []
    chapter_num = 0
    for chapter in chapters:
        chapter_num += 1
        doc = nlp(chapter)
        sentence_num = 0
        for sent in doc.sents:
            sentence_num += 1
            for token in sent:
                if token.is_alpha:
                    rows.append({"word": token.text.lower(),
                                 "lemma": token.lemma_.lower(),
                                 "pos": token.pos_,
                                 "is_stop": token.is_stop,
                                 "sentence": sentence_num,
                                 "chapter": chapter_num})
    df = pd.DataFrame(rows)
    df["text"] = label
    return df

def book_to_svo_df(filename, label, nlp):
    text = load_text(filename)
    chapters = re.split(r"(?im)^CHAPTER\s+(?:[IVXLCDM]+|\d+)\b", text)
    chapters = chapters[1:]
    rows = []
    chapter_num = 0
    for chapter in chapters:
        chapter_num += 1
        doc = nlp(chapter)
        for token in doc:
            if token.pos_ == "VERB":
                subjects = [c for c in token.children if c.dep_ == "nsubj"]
                objects = [c for c in token.children if c.dep_ == "dobj"]
                if subjects and objects:
                    rows.append({"subject": subjects[0].lemma_.lower(),
                                 "verb": token.lemma_.lower(),
                                 "object": objects[0].lemma_.lower(),
                                 "chapter": chapter_num})
    df = pd.DataFrame(rows)
    df["text"] = label
    return df

What Is WordNet?

WordNet

  • A lexical database of English, built at Princeton University
  • Groups words into synsets: sets of words that share one meaning
  • Connects synsets to each other:
    • cat is a kind of feline, which is a kind of mammal… (hyponyms)
    • wheel is a part of car (meronyms)
    • fast is the opposite of slow (antonyms)

Going Up: Hypernyms

What Kind of Thing Is It?

  • A hypernym is a more general word: soup is a kind of dish.
soup = wn.synset("soup.n.01")
soup.hypernyms()
[Synset('dish.n.02')]
  • And the reverse, a more specific word, is a hyponym: soup is a hyponym of dish.

All the Way Up

for path in soup.hypernym_paths():

Is It Food?

Two Kinds of Food

print(wn.synset("food.n.01").definition())
print(wn.synset("food.n.02").definition())
any substance that can be metabolized by an animal to give energy and build tissue
any solid substance (as opposed to liquid) that is used as a source of nourishment
  • soup is food.n.01; cake is food.n.02.
  • food.n.01 is the broader one – but Alice’s cake (cake.n.03) isn’t under it!

A Function: is_a()

  • Is any noun sense of a word under a category?
def is_a(word, category):
    for synset in wn.synsets(word, pos=wn.NOUN):
        for path in synset.hypernym_paths():
            if category in path:
                return True
    return False
food = wn.synset("food.n.01")
print(is_a("soup", food))
print(is_a("hatter", food))
True
False

Other Relations

Opposites: Antonyms

  • Antonyms belong to lemmas (words), not synsets.
antonyms = []
for synset in wn.synsets("slow"):
    for lemma in synset.lemmas():
        for ant in lemma.antonyms():
            antonyms.append(ant.name())
print(antonyms)
['accelerate', 'fast', 'fast', 'fast', 'quickly']

Parts and Wholes: Meronyms

wn.synset("car.n.01").part_meronyms()[:6]
[Synset('car_mirror.n.01'), Synset('first_gear.n.01'), Synset('car_door.n.01'),
 Synset('reverse.n.02'), Synset('stabilizer_bar.n.01'), Synset('bumper.n.02')]
  • A meronym is a part: a car door is part of a car.
  • The opposite, part_holonyms(), gives the whole.

Verbs: Entailments

print(wn.synset("snore.v.01").entailments())
print(wn.synset("buy.v.01").entailments())
[Synset('sleep.v.01')]
[Synset('choose.v.01'), Synset('pay.v.01')]
  • If you snore, you must be sleeping. If you buy, you choose and pay.

How Similar? Wu–Palmer Similarity

  • How close are two synsets in the tree? From 0 (unrelated) to 1 (the same).
dog = wn.synset("dog.n.01")
for other in ["cat.n.01", "rabbit.n.01", "bat.n.01", "cake.n.03"]:
    print(other, dog.wup_similarity(wn.synset(other)))
cat.n.01 0.8571428571428571
rabbit.n.01 0.7857142857142857
bat.n.01 0.8461538461538461
cake.n.03 0.25
  • The closer their shared ancestor is to them, the higher the score.
