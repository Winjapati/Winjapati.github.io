import os
import re
import pandas as pd
from nltk.corpus import wordnet as wn

HERE = os.path.dirname(__file__)   # the folder lin301.py lives in: data/

def ttr(words):
    return len(set(words)) / len(words)

def ttr_first(words, n=10000):
    return len(set(words[:n])) / len(words[:n])

def avg_word_len(words):
    total = 0
    for w in words:
        total += len(w)
    return total / len(words)

def ly_rate(words):
    count = 0
    for w in words:
        if w.endswith("ly"):
            count += 1
    return count / len(words) * 1000

def stopword_rate(words, stopwords):
    count = 0
    for w in words:
        if w in stopwords:
            count += 1
    return count / len(words)

def load_text(filename):
    with open(os.path.join(HERE, "gutenberg", filename), "r", encoding="utf-8") as f:
        lines = f.readlines()
    while not lines[0].startswith("*** START"):
        lines = lines[1:]
    lines = lines[1:]
    while not lines[-1].startswith("*** END"):
        lines = lines[:-1]
    lines = lines[:-1]
    return "".join(lines)

def load_words(filename):
    text = load_text(filename)
    words = re.split(r"[\W_]+", text.lower())
    return [w for w in words if w != ""]

def report(title, words):
    print(title, "TTR:", ttr(words))
    print(title, "TTR (10,000 words):", ttr_first(words))
    print(title, "Average Word Length:", avg_word_len(words))
    print(title, "-ly words per 1,000:", ly_rate(words))
    return ttr_first(words)

def book_to_df(filename, label):
    text = load_text(filename)
    chapters = re.split(r"(?im)^CHAPTER\s+(?:[IVXLCDM]+|\d+)\b", text)
    chapters = chapters[1:]
    rows = []
    chapter_num = 0
    for chapter in chapters:
        chapter_num += 1
        words = re.split(r"[\W_]+", chapter.lower())
        words = [w for w in words if w != ""]
        for w in words:
            rows.append({"word": w, "chapter": chapter_num})
    df = pd.DataFrame(rows)
    df["text"] = label
    return df

def add_columns(df):
    df["length"] = df["word"].str.len()
    df["is_ly"] = df["word"].str.endswith("ly")
    return df

def book_to_spacy_df(filename, label, nlp):
    text = load_text(filename)
    chapters = re.split(r"(?im)^CHAPTER\s+(?:[IVXLCDM]+|\d+)\b", text)
    chapters = chapters[1:]
    rows = []
    chapter_num = 0
    for chapter in chapters:
        chapter_num += 1
        doc = nlp(chapter)
        sentence_num = 0
        for sent in doc.sents:
            sentence_num += 1
            for token in sent:
                if token.is_alpha:
                    rows.append({"word": token.text.lower(),
                                 "lemma": token.lemma_.lower(),
                                 "pos": token.pos_,
                                 "is_stop": token.is_stop,
                                 "sentence": sentence_num,
                                 "chapter": chapter_num})
    df = pd.DataFrame(rows)
    df["text"] = label
    return df

def book_to_svo_df(filename, label, nlp):
    text = load_text(filename)
    chapters = re.split(r"(?im)^CHAPTER\s+(?:[IVXLCDM]+|\d+)\b", text)
    chapters = chapters[1:]
    rows = []
    chapter_num = 0
    for chapter in chapters:
        chapter_num += 1
        doc = nlp(chapter)
        for token in doc:
            if token.pos_ == "VERB":
                subjects = [c for c in token.children if c.dep_ == "nsubj"]
                objects = [c for c in token.children if c.dep_ == "dobj"]
                if subjects and objects:
                    rows.append({"subject": subjects[0].lemma_.lower(),
                                 "verb": token.lemma_.lower(),
                                 "object": objects[0].lemma_.lower(),
                                 "chapter": chapter_num})
    df = pd.DataFrame(rows)
    df["text"] = label
    return df

def is_a(word, category):
    for synset in wn.synsets(word, pos=wn.NOUN):
        for path in synset.hypernym_paths():
            if category in path:
                return True
    return False

Text Is Numbers: Unicode

Every Character Is a Number

  • Computers store text as numbers. Unicode gives every character in every writing system its own number.
print(ord("a"))
print(ord("é"))
print(ord("γ"))
print(ord("猫"))
print(chr(233))

  • ord(): character → number; chr(): number → character

Every Character Has a Name

  • Remember the kahakō and ʻokina from HW 6? They’re Unicode characters too.
  • The ʻokina is not an apostrophe ('): it’s a different character, with a different number.

Same Look, Different String

  • é can be written two ways: one character, or e plus a combining accent.
a = "é"
b = "e\u0301"   # e + COMBINING ACUTE ACCENT
print(a, b)
print(len(a), len(b))
print(a == b)
é é
1 2
False
  • They look identical, but Python says they’re different!

Normalizing

print([unicodedata.name(c) for c in b])
print(unicodedata.normalize("NFC", b) == a)
['LATIN SMALL LETTER E', 'COMBINING ACUTE ACCENT']
True
  • "NFC" combines letters and accents into single characters wherever it can.
  • Normalize text from different sources before you compare or count it.

Files Are Bytes: Encodings

What encoding="utf-8" Means

  • A file stores bytes, not characters.
  • An encoding is the rule for turning characters into bytes, and back.
  • UTF-8 is today’s standard: it can store every Unicode character.
  • That’s why we always write open(..., encoding="utf-8").

Older Files, Older Encodings

  • Let’s save a sentence from the Spanish Alice in an older encoding, Latin-1, and then read it back as UTF-8.
with open("spanish_latin1.txt", "w", encoding="latin-1") as f:
    f.write("Alicia, la niña, era matemática.")

with open("spanish_latin1.txt", "r", encoding="utf-8") as f:
    text = f.read()

Older Files, Older Encodings

  • This gives the following error:
UnicodeDecodeError: 'utf-8' codec can't decode byte 0xf1 in position 13:
invalid continuation byte
  • In Latin-1, ñ is one byte (0xf1). UTF-8 doesn’t allow that byte on its own.

Reading It Correctly

with open("spanish_latin1.txt", "r", encoding="latin-1") as f:
    print(f.read())
Alicia, la niña, era matemática.
  • Same file, right encoding: no problem.

The Opposite Mistake

with open("spanish_utf8.txt", "w", encoding="utf-8") as f:
    f.write("Alicia, la niña, era matemática.")

with open("spanish_utf8.txt", "r", encoding="latin-1") as f:
    print(f.read())
Alicia, la niÃ±a, era matemÃ¡tica.
  • No error – just garbled characters! This is called mojibake.
  • If you see Ã± or Ã¡ in a text, it’s UTF-8 read with the wrong encoding.

Same Book, Three Languages

What’s English-Specific in Our Function?

chapters = re.split(r"(?im)^CHAPTER\s+(?:[IVXLCDM]+|\d+)\b", text)
  • French Alice has CHAPITRE PREMIER, CHAPITRE II…
  • Italian Alice has CAPITOLO I, CAPITOLO II…
  • The word for chapter should be a parameter!

A Parameter with a Default

  • Remember ttr_first(words, n=10000)? If you don’t give n, it uses 10000.
  • We’ll do the same: chapter_word="CHAPTER"
def book_to_spacy_df(filename, label, nlp, chapter_word="CHAPTER"):
    text = load_text(filename)
    chapters = re.split(r"(?m)^" + chapter_word + r"\b.*$", text)
  • r"(?m)^" + chapter_word + r"\b.*$": a line that starts with the chapter word, and everything after it on that line
  • English calls still work: book_to_spacy_df("alice.txt", "English", nlp)

Italian Is Pro-Drop

  • English: She was beginning to get tired. (She is required)
  • French: Elle commençait à se fatiguer. (Elle is required)
  • Italian: Cominciava a stancarsi. (no pronoun needed!)
  • The verb ending -a already tells us ‘she/he’, so the subject pronoun can be dropped.
  • Our numbers found a real typological difference!
